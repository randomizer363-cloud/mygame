// Функция для вибрации (обертка для удобства) function triggerHaptic(type) { try { switch(type) { case 'light': tg.HapticFeedback.impactOccurred('light'); break; case 'medium': tg.HapticFeedback.impactOccurred('medium'); break; case 'error': tg.HapticFeedback.notificationOccurred('error'); break; case 'success': tg.HapticFeedback.notificationOccurred('success'); break; case 'warning': tg.HapticFeedback.notificationOccurred('warning'); break; } } catch(e) { console.log("Haptics not supported"); } } function handleChoice(side) { if(!gameActive || isAnimating) return; isAnimating = true; const cardEl = document.getElementById('card'); const config = levelConfig[currentLevel]; cardEl.classList.add(side === 'left' ? 'fly-left' : 'fly-right'); const isCorrect = (side === 'right' && currentItem === 'cutlet') || (side === 'left' && currentItem === 'fly'); if(currentItem === 'ivan') { if(side === beerSide) { points += 5; playSound('snd-success'); triggerHaptic('success'); // Победная вибрация за Ивана } else { triggerHaptic('error'); // Тряска при краже endGame("ivan_stole"); return; } } else if(isCorrect) { timeLeft += 1.0; if(currentItem === 'cutlet') points += 0.1; playSound('snd-success'); triggerHaptic('light'); // Легкий отклик при успехе } else { errorCount++; if(currentItem === 'fly') points = Math.max(0, points - config.flyPenalty); playSound(errorCount % 5 === 0 ? 'snd-error-big' : 'snd-error'); if(errorCount >= 15) { triggerHaptic('error'); // Сильная вибрация при проигрыше endGame("fatal"); } else { triggerHaptic('warning'); // Предупреждение при ошибке if(errorCount % 5 === 0) showMsg("ОШИБКА!"); } } updateLabels(); setTimeout(() => { if (points >= config.target && gameActive) { triggerHaptic('success'); // Вибрация триумфа endGame("win"); } else if (gameActive) { nextItem(); } isAnimating = false; }, 250); } function endGame(reason) { gameActive = false; clearInterval(gameLoop); let title = "ИГРА ОКОНЧЕНА", msg = ""; if(reason === "win") { title = "УРОВЕНЬ ПРОЙДЕН! 🎉"; triggerHaptic('success'); // Финальный аккорд победы if(currentLevel === unlockedLevels) unlockedLevels++; localStorage.setItem('unlockedLevels', unlockedLevels); playSound('snd-win'); } else { triggerHaptic('error'); // Финальный аккорд поражения msg = reason === "ivan_stole" ? "Иван украл котлеты!" : (reason === "fatal" ? "СМЕРТЕЛЬНО: 15 ОШИБОК" : "ВРЕМЯ ВЫШЛО"); playSound('snd-lose'); } document.getElementById('finish-content').innerHTML = `

${title}

${msg}

Время: ${totalTime.toFixed(1)}с

`; document.getElementById('finish-overlay').style.display = 'flex'; }
